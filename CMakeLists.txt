if(TELA_BUILD_GPU)
  # Pictor's UIRenderer and VulkanContext exist only when Pictor was built with PICTOR_HAS_VULKAN.
  find_package(Vulkan REQUIRED)
  add_library(tela_gpu src/infrastructure/gpu/gpu_renderer.cpp src/infrastructure/gpu/vulkan_layers.cpp
    src/infrastructure/gpu/ui_commands.cpp src/infrastructure/gpu/layer_raster.cpp)
  add_library(Tela::Gpu ALIAS tela_gpu)
  target_include_directories(tela_gpu PRIVATE ${TELA_PICTOR_INCLUDE_DIR})
  target_compile_definitions(tela_gpu PRIVATE PICTOR_HAS_VULKAN NOMINMAX)
  target_link_libraries(tela_gpu PUBLIC Tela::GpuFrame Tela::Pictor Vulkan::Vulkan PRIVATE TelaPictorDependency)
  set_target_properties(tela_gpu PROPERTIES EXPORT_NAME Gpu)
  install(TARGETS tela_gpu EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  find_path(TELA_GLFW_INCLUDE_DIR GLFW/glfw3.h HINTS ${TELA_PICTOR_INCLUDE_DIR} REQUIRED)
  find_library(TELA_GLFW_LIBRARY NAMES glfw3 glfw REQUIRED)
  add_executable(tela_gpu_controls examples/gpu/gpu_controls_host.cpp examples/gpu/glfw_input.cpp
    examples/controls/controls_gallery.cpp)
  target_include_directories(tela_gpu_controls PRIVATE ${TELA_PICTOR_INCLUDE_DIR} ${TELA_GLFW_INCLUDE_DIR})
  target_compile_definitions(tela_gpu_controls PRIVATE PICTOR_HAS_VULKAN NOMINMAX)
  target_link_libraries(tela_gpu_controls PRIVATE Tela::Gpu TelaPictorDependency ${TELA_GLFW_LIBRARY} Vulkan::Vulkan)
  if(WIN32)
    target_link_libraries(tela_gpu_controls PRIVATE user32 gdi32 shell32)
  endif()
  if(TELA_PICTOR_SHADER_SOURCE_DIR)
    find_program(TELA_GLSLC glslc HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/bin" REQUIRED)
    set(_tela_shader_out ${CMAKE_CURRENT_BINARY_DIR}/shaders)
    set(_tela_spirv)
    foreach(_stage vert frag)
      set(_spv ${_tela_shader_out}/ui_widget.${_stage}.spv)
      add_custom_command(OUTPUT ${_spv}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${_tela_shader_out}
        COMMAND ${TELA_GLSLC} ${TELA_PICTOR_SHADER_SOURCE_DIR}/ui_widget.${_stage} -o ${_spv}
        DEPENDS ${TELA_PICTOR_SHADER_SOURCE_DIR}/ui_widget.${_stage} VERBATIM)
      list(APPEND _tela_spirv ${_spv})
    endforeach()
    add_custom_target(tela_gpu_shaders DEPENDS ${_tela_spirv})
    add_dependencies(tela_gpu_controls tela_gpu_shaders)
    target_compile_definitions(tela_gpu_controls PRIVATE TELA_GPU_SHADER_DIR="${_tela_shader_out}")
  endif()
endif()
