cmake_minimum_required(VERSION 3.20)
project(Tela VERSION 0.1.0 LANGUAGES CXX)
add_library(tela_core src/domain/composition/document.cpp src/domain/composition/layout.cpp
  src/application/runtime.cpp src/domain/input/input_regions.cpp src/domain/input/gesture_ownership.cpp
  src/infrastructure/bridge/bridge_codec.cpp src/application/bridge_session.cpp
  src/domain/overlay/presentation_status.cpp)
add_library(Tela::Core ALIAS tela_core)
include(GNUInstallDirs)
target_include_directories(tela_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(tela_core PUBLIC cxx_std_20)
set_target_properties(tela_core PROPERTIES EXPORT_NAME Core)
add_library(tela_transitions src/application/transitions.cpp src/application/transition_document.cpp)
add_library(Tela::Transitions ALIAS tela_transitions)
target_link_libraries(tela_transitions PUBLIC Tela::Core)
set_target_properties(tela_transitions PROPERTIES EXPORT_NAME Transitions)
option(TELA_BUILD_WINDOWS "Build the explicit Pictor CPU/Windows composition adapter" OFF)
if(TELA_BUILD_WINDOWS)
  if(NOT WIN32)
    message(FATAL_ERROR "TELA_BUILD_WINDOWS requires Windows")
  endif()
  find_path(TELA_PICTOR_INCLUDE_DIR pictor/text/text_image_renderer.h REQUIRED)
  find_library(TELA_PICTOR_LIBRARY NAMES pictor REQUIRED)
  add_library(TelaPictorDependency UNKNOWN IMPORTED)
  set_target_properties(TelaPictorDependency PROPERTIES IMPORTED_LOCATION "${TELA_PICTOR_LIBRARY}")
  add_library(tela_pictor src/infrastructure/pictor/pictor_surface.cpp)
  add_library(Tela::Pictor ALIAS tela_pictor)
  target_include_directories(tela_pictor PRIVATE ${TELA_PICTOR_INCLUDE_DIR})
  target_link_libraries(tela_pictor PUBLIC Tela::Core PRIVATE TelaPictorDependency)
  set_target_properties(tela_pictor PROPERTIES EXPORT_NAME Pictor)
  add_library(tela_windows src/infrastructure/windows/layered_bitmap.cpp src/infrastructure/windows/windows_overlay.cpp
    src/infrastructure/windows/pipe_security.cpp src/infrastructure/windows/windows_pipe.cpp)
  add_library(Tela::Windows ALIAS tela_windows)
  target_link_libraries(tela_windows PUBLIC Tela::Pictor PRIVATE user32 gdi32 advapi32)
  target_compile_definitions(tela_windows PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  set_target_properties(tela_windows PROPERTIES EXPORT_NAME Windows)
  add_executable(tela_overlay examples/windows/overlay_host.cpp examples/windows/transition_editor.cpp
    examples/windows/overlay_options.cpp examples/windows/overlay_probe.cpp
    examples/windows/overlay_diagnostics.cpp examples/windows/overlay_session.cpp)
  target_link_libraries(tela_overlay PRIVATE Tela::Windows Tela::Transitions)
  target_compile_definitions(tela_overlay PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  add_executable(tela_probe_target examples/windows/probe_target.cpp)
  target_link_libraries(tela_probe_target PRIVATE user32 gdi32)
  target_compile_definitions(tela_probe_target PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  install(TARGETS tela_pictor tela_windows EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(TARGETS tela_overlay RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_executable(tela_declaration examples/declaration.cpp)
target_link_libraries(tela_declaration PRIVATE Tela::Core)
include(CTest)
if(BUILD_TESTING)
  add_executable(tela_presentation_tests tests/presentation_test.cpp)
  target_link_libraries(tela_presentation_tests PRIVATE Tela::Core)
  add_test(NAME tela_presentation_reasons COMMAND tela_presentation_tests)
  add_executable(tela_reconnect_tests tests/reconnect_test.cpp)
  target_link_libraries(tela_reconnect_tests PRIVATE Tela::Core)
  add_test(NAME tela_reconnect_lifetime COMMAND tela_reconnect_tests)
  add_executable(tela_core_tests tests/document_test.cpp)
  target_link_libraries(tela_core_tests PRIVATE Tela::Core)
  add_test(NAME tela_core_contract COMMAND tela_core_tests)
  add_executable(tela_runtime_tests tests/runtime_test.cpp)
  target_link_libraries(tela_runtime_tests PRIVATE Tela::Core)
  add_test(NAME tela_runtime_contract COMMAND tela_runtime_tests)
  add_executable(tela_bridge_tests tests/bridge_test.cpp)
  target_link_libraries(tela_bridge_tests PRIVATE Tela::Core)
  add_test(NAME tela_bridge_contract COMMAND tela_bridge_tests)
  add_executable(tela_transition_tests tests/transitions_test.cpp)
  target_link_libraries(tela_transition_tests PRIVATE Tela::Transitions)
  add_test(NAME tela_transition_persistence COMMAND tela_transition_tests ${CMAKE_CURRENT_BINARY_DIR}/transition-test.tela)
endif()
install(TARGETS tela_core tela_transitions EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/tela DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT TelaTargets NAMESPACE Tela:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tela)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/TelaConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/TelaConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tela)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TelaConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TelaConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/TelaConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Tela)
