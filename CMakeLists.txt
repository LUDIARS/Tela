cmake_minimum_required(VERSION 3.20)
project(Tela VERSION 0.1.0 LANGUAGES CXX)
add_library(tela_core src/domain/composition/document.cpp src/domain/composition/layout.cpp
  src/domain/composition/drawing.cpp src/domain/composition/callout.cpp
  src/application/runtime.cpp src/domain/input/input_regions.cpp src/domain/input/gesture_ownership.cpp
  src/infrastructure/bridge/bridge_codec.cpp src/application/bridge_session.cpp
  src/domain/overlay/presentation_status.cpp src/domain/overlay/placement.cpp)

include(GNUInstallDirs)

  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(tela_core PUBLIC cxx_std_20)
if(MSVC)
  # Exception paths own files, windows and renderer resources; enable C++ unwinding.
  target_compile_options(tela_core PUBLIC /EHsc /utf-8)
endif()
set_target_properties(tela_core PROPERTIES EXPORT_NAME Core)

option(TELA_BUILD_WINDOWS "Build the explicit Pictor CPU/Windows composition adapter" OFF)
option(TELA_BUILD_MACOS "Build the explicit Pictor CPU/AppKit composition adapter" OFF)
option(TELA_BUILD_PICTOR "Build the portable Pictor CPU surface adapter" OFF)
if(TELA_BUILD_WINDOWS AND NOT WIN32)
  message(FATAL_ERROR "TELA_BUILD_WINDOWS requires Windows")
endif()
if(TELA_BUILD_MACOS AND NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "TELA_BUILD_MACOS requires macOS")
endif()
if(TELA_BUILD_WINDOWS OR TELA_BUILD_MACOS OR TELA_BUILD_PICTOR)
  find_path(TELA_PICTOR_INCLUDE_DIR pictor/text/text_image_renderer.h REQUIRED)
  find_library(TELA_PICTOR_LIBRARY NAMES pictor REQUIRED)
  add_library(TelaPictorDependency UNKNOWN IMPORTED)
  set_target_properties(TelaPictorDependency PROPERTIES IMPORTED_LOCATION "${TELA_PICTOR_LIBRARY}")
  add_library(tela_pictor src/infrastructure/pictor/pictor_surface.cpp)
  add_library(Tela::Pictor ALIAS tela_pictor)
  target_include_directories(tela_pictor PRIVATE ${TELA_PICTOR_INCLUDE_DIR})
  target_link_libraries(tela_pictor PUBLIC Tela::Core Tela::Raster PRIVATE TelaPictorDependency)
  set_target_properties(tela_pictor PROPERTIES EXPORT_NAME Pictor)
  install(TARGETS tela_pictor EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(TELA_BUILD_MACOS)
  enable_language(OBJCXX)
  add_library(tela_macos src/infrastructure/macos/macos_overlay.mm
    src/infrastructure/macos/window_target.mm src/infrastructure/macos/bitmap_view.mm
    src/infrastructure/macos/input_panel.mm src/infrastructure/macos/region_surface.mm)
  add_library(Tela::MacOS ALIAS tela_macos)
  set_target_properties(tela_macos PROPERTIES EXPORT_NAME MacOS OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(tela_macos PRIVATE -fobjc-arc)
  target_link_libraries(tela_macos PUBLIC Tela::Pictor PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework QuartzCore")
  add_executable(tela_macos_overlay examples/macos/overlay_host.mm)
  set_target_properties(tela_macos_overlay PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(tela_macos_overlay PRIVATE -fobjc-arc)
  target_link_libraries(tela_macos_overlay PRIVATE Tela::MacOS "-framework AppKit")
  install(TARGETS tela_macos EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(TARGETS tela_macos_overlay RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TELA_BUILD_WINDOWS)
  add_library(tela_windows src/infrastructure/windows/layered_bitmap.cpp src/infrastructure/windows/windows_overlay.cpp

    examples/windows/graph_content.cpp examples/windows/attached_viewport.cpp)

  target_compile_definitions(tela_overlay PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  add_executable(tela_view examples/windows/view_host.cpp examples/windows/view_options.cpp
    examples/windows/spec_view_content.cpp examples/windows/graph_content.cpp)
  target_link_libraries(tela_view PRIVATE Tela::Windows Tela::SpecView Tela::Graph)
  target_compile_definitions(tela_view PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  install(TARGETS tela_view RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  add_executable(tela_source examples/windows/source_host.cpp examples/windows/source_options.cpp
    examples/windows/source_content.cpp)
  target_link_libraries(tela_source PRIVATE Tela::Windows Tela::SourceView)

  add_executable(tela_probe_target examples/windows/probe_target.cpp)
  target_link_libraries(tela_probe_target PRIVATE user32 gdi32)
  target_compile_definitions(tela_probe_target PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  install(TARGETS tela_windows EXPORT TelaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(TARGETS tela_overlay RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
